// ── Damage Formula (CalcDamage) ─────────────────
float base = Max(1, atk - Floor(def × 0.5));
float crit = Clamp(lck × 0.5f, 0, 25) / 100f;
bool isCrit = random < crit;
float v = 1 ± 10% variance;
int dmg = Floor(base × v × (isCrit ? 2 : 1));
 
// ── EXP Curve ──────────────────────────────────
expNeeded = Floor(100 × 1.45^(level - 1));
 
// ── Level-Up Stat Growth ────────────────────────
maxHp += Floor(5 + con × 0.4 + rand × 4);
maxMp += Floor(2 + int × 0.25 + rand × 3);
atk += Floor(0.3 + str × 0.05);
def += Floor(0.2 + con × 0.04);
str++; con++; int++;
if (rand > 0.7) lck++; // luck: rare growth
hp = maxHp; mp = maxMp; // full heal
